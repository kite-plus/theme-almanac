{{- /* The header links, from the nav setting, for the header itself and for the menu of a narrow screen. Called with (dict "ctx" . "variant" "header" or "drawer"). With no links set, the home page and the archive stand in. */ -}} {{- $ctx := .ctx -}} {{- $variant := .variant -}} {{- $lang := $ctx.Site.Language -}} {{- $here := $ctx.Page.RelPermalink -}} {{- $homeURL := url.For "home" -}} {{- $items := $ctx.Site.ThemeSettings.nav -}} {{- if not $items }} {{- $items = slice (dict "label" (partial "t.html" (dict "k" "posts" "lang" $lang)) "href" (url.For "list" "post") "icon" "file-text") (dict "label" (partial "t.html" (dict "k" "tags" "lang" $lang)) "href" (url.For "taxonomy" "tags") "icon" "hash") }} {{- end -}} {{- $guess := dict "posts" "file-text" "projects" "folder" "douban" "book-open" "reading" "book-open" "watching" "film" "archives" "archive" "moments" "sparkles" "about" "user" "links" "external-link" "resume" "id-card" "tags" "hash" "categories" "folder" "feed" "rss" "search" "search" "travel" "plane" -}} {{- range $items }} {{- if and .label (or .href .url) }} {{- $href := .href }} {{- if not $href }}{{ $href = url.Rel .url }}{{ end }} {{- $external := not (str.HasPrefix "/" $href) }} {{- $active := false }} {{- if $external }} {{- else if eq $href $homeURL }}{{ $active = eq $here $homeURL }} {{- else }}{{ $active = str.HasPrefix $href $here }}{{ end }} {{- if eq $variant "drawer" }} {{- $icon := .icon }} {{- if or (not $icon) (eq $icon "auto") }} {{- $icon = "chevron-right" }} {{- if eq $href $homeURL }}{{ $icon = "rocket" }}{{ end }} {{- range $key, $name := $guess }}{{ if str.Contains (printf "/%s" $key) $href }}{{ $icon = $name }}{{ end }}{{ end }} {{- end }} {{- partial "icon.html" (dict "name" $icon "class" "size-4") -}} {{ .label }} {{- else }} {{ .label }} {{- end }} {{- end }} {{- end -}}
